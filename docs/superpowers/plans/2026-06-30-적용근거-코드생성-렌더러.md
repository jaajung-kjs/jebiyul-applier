## Global Constraints

- Python 3.11 이상. 외부 의존성은 `openpyxl`, `pytest`, `pyinstaller`만.
- 율 값은 소수(0.126)로 다루고 셀 표기는 number_format(`0.0%`, `0.00%`, `0.000%`)로 한다.

- 양식이 예상과 다르면 조용히 틀린 값을 내지 말고 예외를 던진다.
- 골든/실파일 테스트는 제비율 원본 파일(`붙임1…xlsx`, `붙임2…xlsx`)이 리포지토리 루트에 있을 때만 실행하고 없으면 skip한다(기존 정책).
- PIU 추출 스타일 값(고정):
  - 본문 폰트 `맑은 고딕` 11; 시트 제목 폰트 `HY견고딕` 24 center.
  - 섹션 제목(1./2.…): 본문 폰트 11 **bold**, 글자색 `FF0000FF`(파랑).
  - 적용율 라벨/값: 본문 폰트 11 **bold**, 글자색 `FFFF0000`(빨강).
  - 표 헤더 채움: solid `FFDBE5F1`(accent1 80% lighter).
  - 적용행 강조 채움(신규): solid `FFFFF2CC`(옅은 금색).

```python
# src/styles.py
"""PIU 8.적용근거에서 추출한 스타일 팔레트. 렌더러가 셀에 입히는 폰트·채움·테두리·정렬·치수."""
from openpyxl.styles import Font, PatternFill, Border, Side, Alignment

BODY_FONT_NAME = "맑은 고딕"
TITLE_FONT_NAME = "HY견고딕"

HEADER_FILL_RGB = "FFDBE5F1"      # accent1 80% lighter
HIGHLIGHT_FILL_RGB = "FFFFF2CC"   # 적용행 강조(옅은 금색)
SECTION_COLOR = "FF0000FF"        # 섹션 제목(파랑)
RATE_COLOR = "FFFF0000"           # 적용율(빨강)

COL_WIDTHS = {"A": 4.2, "B": 10.2, "C": 14.1, "D": 10.2, "E": 12.6,
              "G": 10.2, "H": 12.4, "I": 10.2, "K": 2.5}
ROW_TITLE_H = 35.1
ROW_BODY_H = 20.1


def body_font():
    return Font(name=BODY_FONT_NAME, size=11)


def title_font():
    return Font(name=TITLE_FONT_NAME, size=24)


def section_font():
    return Font(name=BODY_FONT_NAME, size=11, bold=True, color=SECTION_COLOR)


def rate_font():
    return Font(name=BODY_FONT_NAME, size=11, bold=True, color=RATE_COLOR)


def header_fill():
    return PatternFill(patternType="solid", fgColor=HEADER_FILL_RGB)


def highlight_fill():
    return PatternFill(patternType="solid", fgColor=HIGHLIGHT_FILL_RGB)


def side(style):
    return Side(style=style)


def box_border(outer="thin", inner="hair"):
    return Border(top=side(outer), bottom=side(outer),
                  left=side(outer), right=side(outer))


def center(wrap=False):
    return Alignment(horizontal="center", vertical="center", wrap_text=wrap)


def left():
    return Alignment(horizontal="left", vertical="center")


def right():
    return Alignment(horizontal="right", vertical="center")
```

```python
블록 리스트를 위→아래로 그릴 뿐 분기 판단을 하지 않는다.

from dataclasses import dataclass, field

NOTE = "[조달청 공사 원가계산 제비율 변경 '24.6.27. 적용]"


@dataclass
class Title:
    text: str


@dataclass
class SectionHeader:
    text: str
    note: str | None = None


@dataclass
class SubHeader:
    text: str


@dataclass
class NoteLines:
    lines: list


@dataclass
class AppliedRate:
    label: str
    value: float
    fmt: str = "0.0%"


@dataclass
class BandRow:
    cells: list
    highlight: bool = False


@dataclass
class BandTable:
    """Task 3~5에서 렌더 규칙과 함께 채운다."""
    kind: str                 # 'gibon' | 'sanan' | 'ilban' | 'iyun'

    rows: list = field(default_factory=list)


def pct(rate: float) -> str:
    """0.03626 → '3.626' (불필요한 0 제거)."""
    return f"{rate * 100:g}"
```

```python
# src/renderer.py 의 _emit_table 교체
def _emit_table(ws, block, cur):
    # 헤더행
    hr = cur.take(1)
    ws.row_dimensions[hr].height = styles.ROW_BODY_H
    for text, c0, c1 in block.headers:
        _put(ws, hr, c0, c1, text, fill=styles.header_fill(),
             align=styles.center(wrap=True), border=True)
    # 데이터행
    for brow in block.rows:
        r = cur.take(1)
        ws.row_dimensions[r].height = styles.ROW_BODY_H
        fill = styles.highlight_fill() if brow.highlight else None
        for cell in brow.cells:
            text, c0, c1, fmt = cell
            _put(ws, r, c0, c1, text, fill=fill, align=styles.center(),
                 border=True, fmt=fmt)


def _put(ws, row, c0, c1, value, fill=None, align=None, border=False, fmt=None):
    a = f"{get_column_letter(c0)}{row}"
    if c1 > c0:
        ws.merge_cells(f"{a}:{get_column_letter(c1)}{row}")
    c = ws[a]
    c.value = value
    c.font = styles.body_font()
    if align: c.alignment = align
    if fill: c.fill = fill
    if fmt: c.number_format = fmt
    if border:
        b = styles.box_border()
        for col in range(c0, c1 + 1):
            ws.cell(row, col).border = b
```

```python
_DURS = [("6개월 이하 (183일)", "183"), ("7-12개월 (365일)", "365"),
         ("13-36개월 (1095일)", "1095"), ("36개월 이상 (1096일)", "1096+")]
_SIZE_LABEL = {"10억미만": "50억 미만", "10-50억": "50억 미만",
               "50-300억": "50억 ~ 300억 미만", "300-1000억": "300억 ~ 1000억 미만",
               "1000억이상": "1000억 이상"}


def _gibon_table(path, item, kind, jikjeop, days):
    size = P.size_band(jikjeop)
    dur_applied = P.duration_band(days)
    is_etc = (item == "기타경비")
    under50 = size in ("10억미만", "10-50억")
    band = "10억미만" if under50 else size      # <50억은 50억미만 스케줄
    rows = []
    for i, (dlabel, dur) in enumerate(_DURS):
        rate = lookup.table_rate(path, item, kind, band, dur)
        size_cell = (_SIZE_LABEL[band], 2, 2, None) if i == 0 else None
        if is_etc:
            comp = round(rate * (19.3 + 30.2 + 17.8 + 12.3) / 100, 3)
            value_cells = [(dlabel, 3, 4, None), (rate, 5, 5, "0.0%"), (comp, 6, 6, "0.0%")]
        else:
            value_cells = [(dlabel, 3, 4, None), (rate, 5, 6, "0.0%")]
        cells = [c for c in [size_cell] + value_cells if c]
        rows.append(BandRow(cells, highlight=(dur == dur_applied)))
    if under50:
        rows.append(BandRow([("50억 이상", 2, 2, None),
                             ("조달청 발표자료 참조(공사규모별, 기간별 적용율)", 3, 6, None)]))
    if is_etc:
        headers = [("공사규모", 2, 2), ("공사기간", 3, 4), ("기준", 5, 5), ("적용율", 6, 6), ("적  용  기  준", 7, 10)]
    else:
        headers = [("공사규모", 2, 2), ("공사기간", 3, 4), ("적용율", 5, 6), ("적  용  기  준", 7, 10)]
    return BandTable(kind="gibon", headers=headers, rows=rows)
```

```python
ITEMS = [
    "간접노무비", "공구손료", "산재보험료", "고용보험료", "건강보험료", "연금보험료",
    "퇴직공제부금비", "노인장기요양보험료", "산업안전보건관리비", "기타경비", "일반관리비", "이윤",
]


def build_output(rates, out_path, params=None, jebiyul_path=None, template_path=None):
    """rates(+선택 params/jebiyul_path)로 적용근거 xlsx를 생성한다.

    template_path 인자는 과거 호환용이며 무시된다(템플릿 복사 폐지).
    """
    blocks = sheet_model.build(params or _params_from_rates(rates), rates, jebiyul_path)
    return renderer.render(blocks, out_path)


def _params_from_rates(rates):
    # params 없이 호출되는 경로(골격만): 표 생략에 필요한 최소 키.
    return {"kind": "", "jikjeop_cost": 0, "days": 0,
```
